## 4. ארכיטקטורת ה-UI

### פריסה: 3 פאנלים ניתנים לגרירה

```
┌─────────────────────────────────────────────────────────────┐
│  עיון הפרשה                                            [⚙]  │
├──────────────────┬───────────────────┬──────────────────────┤
│                  │                   │                      │
│  שאלות           │  צורת הדף          │  עורך תשובות         │

│                  │                   │                      │
│  עמודה ימנית     │  עמודה אמצעית      │  עמודה שמאלית        │
│  ~30%            │  ~40%             │  ~30%                │
│                  │                   │                      │
└──────────────────┴───────────────────┴──────────────────────┘
```

### עקרונות UI — חובה לעמוד בכולם

- **CSS variables בלבד** — אסור להשתמש בצבעים מקודדים. כל צבע נשלף מ-`app.getTheme()`.
- **Frank Ruhl Libre** — פונט ברירת מחדל לכל הטקסטים העבריים.
- **RTL מלא** — `dir="rtl"` על `<html>`, `<body>`, וכל פאנל. סרגלי כלים מיושרים מימין.
- **Dark/Light mode** — מנוי על `theme.changed` ועדכון CSS variables בזמן אמת.
- **Splitters** — `<div class="splitter">` בין הפאנלים, גרירה עם `mousedown/mousemove/mouseup`.
- **Responsive** — הפאנלים מתכווצים לרוחב מינימלי של 200px.

## 5. מחזור החיים של התוסף

```javascript
// app.js — אתחול

window.addEventListener('message', async (event) => {
  if (event.data?.type === 'plugin.boot') {
    const { theme, appVersion, permissions } = event.data.payload;
    applyTheme(theme);
    await initPdfPanel();
    await initLibraryPanel();
    initEditorPanel();
  }

  if (event.data?.type === 'theme.changed') {
    applyTheme(event.data.payload);
  }
});

function applyTheme(theme) {
  const root = document.documentElement;
  root.style.setProperty('--color-primary',    theme.primary);
  root.style.setProperty('--color-secondary',  theme.secondary);
  root.style.setProperty('--color-surface',    theme.surface);
  root.style.setProperty('--color-background', theme.background);
  root.style.setProperty('--color-on-surface', theme.onSurface);
  root.style.setProperty('--color-outline',    theme.outline);
  root.style.setProperty('--color-error',      theme.error);
  root.style.setProperty('--font-family',      theme.fontFamily ?? 'Frank Ruhl Libre, serif');
  root.style.setProperty('--font-size-base',   theme.baseFontSize ?? '16px');
}
```

### 6.2 — pdf-panel.js

```javascript
// pdf-panel.js

let pdfDoc = null;
let currentPage = 1;

async function initPdfPanel() {
  // קבלת מיקום קובץ ה-PDF מ-storage
  const storedPath = await Otzaria.call('plugin.storage.read', { key: 'pdfPath' });

  if (!storedPath?.value) {
    showPdfUploadPrompt();
    return;
  }

  await loadPdf(storedPath.value);
  await navigateToCurrentParasha();
}

async function navigateToCurrentParasha() {
  try {
    // קריאה ל-API לוח השנה של אוצריא
    const result = await Otzaria.call('calendar.getCurrentParasha', {});
    const parashaName = result?.parasha ?? '';

    const index = await fetchParashaIndex();
    const mapping = index.mappings[parashaName];

    if (mapping) {
      goToPage(mapping.startPage);
    } else {
      // fallback: פרשות מחוברות — ניסיון עם חלק ראשון
      const firstPart = parashaName.split('-')[0].trim();
      const fallbackMapping = index.mappings[firstPart];
      goToPage(fallbackMapping?.startPage ?? index.fallbackPage ?? 1);

      if (!fallbackMapping) {
        await Otzaria.call('notifications.send', {
          message: `לא נמצא מיפוי לפרשה: "${parashaName}". הועבר לעמוד הראשון.`,
          type: 'warning'
        });
      }
    }
  } catch (err) {
    console.error('שגיאה בזיהוי פרשת השבוע:', err);
  }
}

async function fetchParashaIndex() {
  const response = await fetch('./data/parasha_index.json');
  return response.json();
}

function goToPage(pageNum) {
  if (!pdfDoc) return;
  currentPage = Math.max(1, Math.min(pageNum, pdfDoc.numPages));
  renderPage(currentPage);
  updatePageDisplay();
}
```

## 7. פאנל ב': צורת הדף (מאגר ספרים)

```javascript
// library-panel.js

async function initLibraryPanel() {
  // חיפוש ספרי הבסיס הנדרשים
  const books = await Otzaria.call('library.findBooks', {
    query: 'שולחן ערוך אורח חיים',
    limit: 1
  });

  if (!books?.results?.length) {
    showBookNotFoundMessage('שולחן ערוך אורח חיים');
    return;
  }

  const bookId = books.results[0].id;
  renderBookView(bookId);
}

async function renderBookView(bookId) {
  const content = await Otzaria.call('library.getBookContent', {
    bookId,
    linkedCommentaries: [
      'משנה ברורה',
      'ביאור הלכה',
      'שער הציון'
    ],
    displayMode: 'page-layout'   // צורת הדף המובנית
  });

  const panel = document.getElementById('library-panel');
  panel.innerHTML = content.html ?? '';
}
```

## 8. פאנל ג': עורך תשובות

```javascript
// editor-panel.js

const AUTOSAVE_INTERVAL_MS = 30_000;
let autosaveTimer = null;

function initEditorPanel() {
  setupEditor();
  loadSavedDraft();
  startAutosave();
}

async function loadSavedDraft() {
  const result = await Otzaria.call('plugin.storage.read', { key: 'currentDraft' });
  if (result?.value) {
    document.getElementById('editor').innerHTML = result.value;
  }
}

async function saveDraft() {
  const content = document.getElementById('editor').innerHTML;
  await Otzaria.call('plugin.storage.write', {
    key: 'currentDraft',
    value: content
  });
}

function startAutosave() {
  autosaveTimer = setInterval(saveDraft, AUTOSAVE_INTERVAL_MS);
}

// ייצוא ל-Word
async function exportToDocx() {
  const content = document.getElementById('editor').innerHTML;
  const parasha = await getCurrentParashaName();
  const filename = `${parasha}_תשובות.docx`;

  // המרה ל-docx באמצעות docx.js או html-to-docx
  const blob = await htmlToDocx(content, { filename, rtl: true });
  downloadBlob(blob, filename);
}

function downloadBlob(blob, filename) {
  const url = URL.createObjectURL(blob);
  const a = document.createElement('a');
  a.href = url;
  a.download = filename;
  a.click();
  URL.revokeObjectURL(url);
}
```
